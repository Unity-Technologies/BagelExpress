## mesh-modifiers | cs | Post-process the text mesh
owner.AddMeshModifier(context =>
{
    var t = Time.realtimeSinceStartup * 5f;
    foreach (var draw in context.draws)
    {
        if (draw.renderType != RenderType.SdfText)
            continue;
        Bob(draw.vertices, t); // 4 vertices per glyph
    }
});

## mesh-modifiers | cs | Move a glyph's 4 vertices together
static void Bob(NativeSlice<Vertex> vertices, float t)
{
    for (int i = 0; i < vertices.Length; i++)
    {
        var x = vertices[i - i % 4].position.x;
        var vertex = vertices[i];
        vertex.position.y += Mathf.Sin(t + x / 18f) * 4f;
        vertices[i] = vertex;
    }
}
